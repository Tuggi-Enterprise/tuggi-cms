┌──────────────────────────────────────────────────────────────────────────────┐
│ 🔐 CONTROLE DE ACESSO                                                        │
└──────────────────────────────────────────────────────────────────────────────┘

  ADMIN
  ├─ ✅ Ver todos os POIs
  ├─ ✅ Editar todos os POIs
  ├─ ✅ Deletar todos os POIs
  ├─ ✅ Criar POI para qualquer cliente
  └─ ✅ Gerenciar todos os clientes

  CLIENT OWNER (role='client', has client_id)
  ├─ ✅ Ver POIs de seu cliente (owner_id = seu client_id)
  ├─ ✅ Editar POIs de seu cliente
  ├─ ✅ Deletar POIs de seu cliente
  ├─ ✅ Criar novos POIs para seu cliente
  ├─ ✅ Linkar outros usuários ao cliente
  └─ ❌ Acessar POIs de outro cliente

  LINKED USER (via client_cms_users)
  ├─ ✅ (owner) Acesso total a POIs do cliente
  ├─ ✅ (manager) Editar e criar POIs
  ├─ ⚠️  (viewer) Apenas leitura
  └─ ❌ Acessar clientes que não está linkado

  EDITOR/VIEWER (não-cliente)
  ├─ ✅ Ver POIs aprovados (público)
  ├─ ✅ Ver clientes registrados
  └─ ❌ Criar/editar/deletar POIs

  ANÔNIMO
  ├─ ✅ Ver POIs aprovados apenas
  └─ ❌ Qualquer operação de escrita

  NOVOS TRIGGERS:
  ───────────────
    core.validate_cms_user_client_id()
    ├─ IF role='client' THEN client_id NOT NULL
    └─ IF role≠'client' THEN client_id = NULL

    core.set_attraction_owner_on_insert()
    └─ ON INSERT: populate owner_id from creator's client_id

  MIGRAÇÃO DE DADOS:
  ──────────────────
    UPDATE attractions SET owner_id = cms_user.client_id
    FROM cms_users
    WHERE created_by = cms_user.id AND cms_user.client_id IS NOT NULL

  interface CmsUser {
    id: string
    email: string
    full_name?: string
    role: 'admin' | 'client' | 'editor' | 'viewer'
    is_active: boolean
    client_id?: string  // ✨ NOVO - apenas para role='client'
    created_at: string
    // ... outros campos
  }

  interface Attraction {
    id: string
    name: string
    city: string
    country: string
    created_by?: string           // ✨ NOVO - quem criou
    owner_id?: string             // ✨ NOVO - qual cliente é dono
    user_id?: string              // ✨ NOVO - alias para created_by
    approved: boolean
    // ... outros campos
  }

  interface POI {
    id: string
    name: string
    city: string
    owner_id?: string             // ✨ NOVO
    created_by?: string           // ✨ NOVO
    approved: boolean
    // ... outros campos
  }

  interface ClientWithUsers extends Client {
    cms_users?: ClientCmsUser[]
    pois_count?: number           // ✨ NOVO
  }
